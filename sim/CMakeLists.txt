add_library(kite_sim src/generator.cpp)
target_include_directories(kite_sim PUBLIC include)
target_link_libraries(kite_sim PUBLIC kite_core)
target_compile_features(kite_sim PUBLIC cxx_std_17)
target_compile_options(kite_sim PRIVATE $<$<CXX_COMPILER_ID:GNU,Clang,AppleClang>:-ffp-contract=off>)
kite_enable_warnings(kite_sim)

if(KITE_BUILD_TESTS)
  add_executable(kite_sim_tests tests/generator_test.cpp)
  target_link_libraries(kite_sim_tests PRIVATE kite_sim kite_core GTest::gmock_main)
  target_compile_definitions(kite_sim_tests PRIVATE
    KITE_SIM_TEST_DATA_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/data")
  kite_enable_warnings(kite_sim_tests)
  gtest_discover_tests(kite_sim_tests)
endif()
